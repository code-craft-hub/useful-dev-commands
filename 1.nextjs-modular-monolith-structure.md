```
┌─────────────────────────────────────────────────────────────────────┐
│                         DEPENDENCY RULES                            │
│                                                                     │
│   app/  ──imports──▶  modules/*  ──imports──▶  shared/             │
│                           │                        │                │
│                           └────────imports─────▶  infrastructure/  │
│                                                                     │
│   ✅ modules/users  CAN   import from  shared/                     │
│   ✅ modules/users  CAN   import from  infrastructure/             │
│   ❌ modules/users  CANNOT import from  modules/products           │
│   ❌ shared/        CANNOT import from  modules/*                  │
│   ❌ infrastructure/ CANNOT import from  modules/*                 │
│                                                                     │
│   Cross-module communication:  Zustand global store OR             │
│                                 Event Bus (custom EventEmitter)     │
└─────────────────────────────────────────────────────────────────────┘
```

## 🔑 TanStack Query Naming Convention

```
modules/users/queries/
│
├── users.queryKeys.ts
│    export const userKeys = {
│      all:    ['users']                      as const,
│      lists:  () => [...userKeys.all, 'list']        as const,
│      list:   (f) => [...userKeys.lists(), { ...f }] as const,
│      detail: (id) => [...userKeys.all, 'detail', id] as const,
│    }
│
├── users.queryOptions.ts
│    export const usersQueryOptions = {
│      list:   (filters) => queryOptions({ queryKey: userKeys.list(filters), queryFn: ... }),
│      detail: (id)      => queryOptions({ queryKey: userKeys.detail(id),   queryFn: ... }),
│    }
│
├── useUsers.query.ts          → export const useUsers = (filters) => useQuery(usersQueryOptions.list(filters))
├── useUser.query.ts           → export const useUser  = (id)      => useQuery(usersQueryOptions.detail(id))
└── useCreateUser.mutation.ts  → export const useCreateUser = () => useMutation({ mutationFn, onSuccess: invalidate })
```
